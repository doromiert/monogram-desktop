cmake_minimum_required(VERSION 3.16)
project(monogram_math LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MATH_DEPS REQUIRED IMPORTED_TARGET pangomm-2.48 cairomm-1.16 tinyxml2 fontconfig freetype2)
if(NOT MICROTEX_SOURCE_DIR OR NOT MONOGRAM_MATH_RESOURCE_DIR)
  message(FATAL_ERROR "Pinned MicroTeX source and installed resource directory are required")
endif()
# Build the pinned Cairo backend against current GTK-family C++ ABIs.
file(READ "${MICROTEX_SOURCE_DIR}/src/platform/cairo/graphic_cairo.cpp" CAIRO_BACKEND)
string(REPLACE "#include <fontconfig/fontconfig.h>" "#include <fontconfig/fontconfig.h>\n#include <fontconfig/fcfreetype.h>" CAIRO_BACKEND "${CAIRO_BACKEND}")
string(REPLACE "Cairo::FORMAT_ARGB32" "Cairo::Surface::Format::ARGB32" CAIRO_BACKEND "${CAIRO_BACKEND}")
foreach(STYLE NORMAL ITALIC)
  string(REPLACE "Pango::STYLE_${STYLE}" "Pango::Style::${STYLE}" CAIRO_BACKEND "${CAIRO_BACKEND}")
endforeach()
foreach(WEIGHT NORMAL BOLD)
  string(REPLACE "Pango::WEIGHT_${WEIGHT}" "Pango::Weight::${WEIGHT}" CAIRO_BACKEND "${CAIRO_BACKEND}")
endforeach()
string(REPLACE "Cairo::LineCap" "Cairo::Context::LineCap" CAIRO_BACKEND "${CAIRO_BACKEND}")
string(REPLACE "Cairo::LineJoin" "Cairo::Context::LineJoin" CAIRO_BACKEND "${CAIRO_BACKEND}")
foreach(CAP BUTT ROUND SQUARE)
  string(REPLACE "Cairo::LINE_CAP_${CAP}" "Cairo::Context::LineCap::${CAP}" CAIRO_BACKEND "${CAIRO_BACKEND}")
endforeach()
foreach(JOIN BEVEL ROUND MITER)
  string(REPLACE "Cairo::LINE_JOIN_${JOIN}" "Cairo::Context::LineJoin::${JOIN}" CAIRO_BACKEND "${CAIRO_BACKEND}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/graphic_cairo.cpp" "${CAIRO_BACKEND}")
file(GLOB_RECURSE MICROTEX_SOURCES CONFIGURE_DEPENDS "${MICROTEX_SOURCE_DIR}/src/*.cpp")
list(FILTER MICROTEX_SOURCES EXCLUDE REGEX "/(samples|platform)/")
add_executable(monogram-math-renderer renderer.cpp ${MICROTEX_SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/graphic_cairo.cpp")
target_include_directories(monogram-math-renderer PRIVATE "${MICROTEX_SOURCE_DIR}/src")
target_compile_definitions(monogram-math-renderer PRIVATE BUILD_GTK NDEBUG MONOGRAM_MATH_RES="${MONOGRAM_MATH_RESOURCE_DIR}")
target_link_libraries(monogram-math-renderer PRIVATE PkgConfig::MATH_DEPS)
install(TARGETS monogram-math-renderer RUNTIME DESTINATION bin)
install(DIRECTORY "${MICROTEX_SOURCE_DIR}/res/" DESTINATION share/monogram-math)
install(FILES "${MICROTEX_SOURCE_DIR}/LICENSE" DESTINATION share/licenses/monogram-math RENAME MicroTeX-MIT.txt)
install(FILES COPYING DESTINATION share/licenses/monogram-math RENAME Monogram-GPL-3.0.txt)
